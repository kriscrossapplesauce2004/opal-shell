cmake_minimum_required(VERSION 3.22)
project(opal_keyboard VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.5 REQUIRED COMPONENTS Core Gui Qml Quick DBus Test)
find_package(LayerShellQt REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(WAYLAND REQUIRED IMPORTED_TARGET wayland-client)
pkg_check_modules(XKB REQUIRED IMPORTED_TARGET xkbcommon)
find_program(WAYLAND_SCANNER wayland-scanner REQUIRED)
set(protocol_sources)
foreach(protocol input-method-unstable-v2 virtual-keyboard-unstable-v1 wlr-virtual-pointer-unstable-v1)
    set(xml ${CMAKE_CURRENT_SOURCE_DIR}/protocols/${protocol}.xml)
    add_custom_command(OUTPUT ${protocol}.h ${protocol}.c
        COMMAND ${WAYLAND_SCANNER} client-header ${xml} ${protocol}.h
        COMMAND ${WAYLAND_SCANNER} private-code ${xml} ${protocol}.c
        DEPENDS ${xml} VERBATIM)
    list(APPEND protocol_sources ${protocol}.c ${protocol}.h)
endforeach()
add_library(opal-input STATIC InputMethod.cpp InputMethod.h Keymap.cpp Keymap.h Gamepad.cpp Gamepad.h PreferenceStore.cpp PreferenceStore.h SuggestionEngine.cpp SuggestionEngine.h PhysicalKeyboard.cpp PhysicalKeyboard.h ClipboardBridge.cpp ClipboardBridge.h ${protocol_sources})
target_include_directories(opal-input PUBLIC ${CMAKE_CURRENT_BINARY_DIR} ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(opal-input PUBLIC Qt6::Core Qt6::Gui Qt6::DBus PkgConfig::WAYLAND PkgConfig::XKB)
target_compile_options(opal-input PRIVATE -Wall -Wextra -Wpedantic)
# Trackpad helper for docked phones: plain C, no Qt.
add_executable(opal-pointer pointer.c wlr-virtual-pointer-unstable-v1.c wlr-virtual-pointer-unstable-v1.h)
target_include_directories(opal-pointer PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
target_link_libraries(opal-pointer PRIVATE PkgConfig::WAYLAND)
target_compile_options(opal-pointer PRIVATE -Wall -Wextra)
qt_add_executable(opal-keyboard main.cpp)
qt_add_resources(opal-keyboard keyboard PREFIX / FILES Main.qml Keyboard.qml Key.qml SuggestionStrip.qml ChoicePanel.qml ClipCard.qml ClipboardPanel.qml DockedBar.qml EditPanel.qml EmojiCell.qml EmojiPanel.qml EmojiResults.qml EmojiSearchBar.qml Glyph.qml HintBar.qml PanelTitle.qml ResizePanel.qml ToolsSheet.qml TrackpadPanel.qml TrackpadSurface.qml EmojiData.js)
target_link_libraries(opal-keyboard PRIVATE opal-input Qt6::Gui Qt6::Qml Qt6::Quick LayerShellQt::Interface)
install(TARGETS opal-keyboard RUNTIME DESTINATION bin)
enable_testing()
add_executable(keymap-test tests/keymap-test.cpp)
target_link_libraries(keymap-test PRIVATE opal-input Qt6::Test)
add_test(NAME keymap COMMAND keymap-test)
add_executable(gamepad-test tests/gamepad-test.cpp)
target_link_libraries(gamepad-test PRIVATE opal-input Qt6::Test)
add_test(NAME gamepad COMMAND gamepad-test)
add_executable(suggestion-test tests/suggestion-test.cpp)
target_link_libraries(suggestion-test PRIVATE opal-input Qt6::Test)
add_test(NAME suggestions COMMAND suggestion-test)
add_executable(devices-test tests/devices-test.cpp)
target_link_libraries(devices-test PRIVATE opal-input Qt6::Test)
add_test(NAME devices COMMAND devices-test)
qt_add_executable(ui-test tests/ui-test.cpp)
qt_add_resources(ui-test keyboard-test PREFIX / FILES Main.qml Keyboard.qml Key.qml SuggestionStrip.qml ChoicePanel.qml ClipCard.qml ClipboardPanel.qml DockedBar.qml EditPanel.qml EmojiCell.qml EmojiPanel.qml EmojiResults.qml EmojiSearchBar.qml Glyph.qml HintBar.qml PanelTitle.qml ResizePanel.qml ToolsSheet.qml TrackpadPanel.qml TrackpadSurface.qml EmojiData.js)
target_link_libraries(ui-test PRIVATE opal-input Qt6::Gui Qt6::Qml Qt6::Quick Qt6::Test)
add_test(NAME keyboard-ui COMMAND ui-test)
set_tests_properties(keyboard-ui PROPERTIES TIMEOUT 120 ENVIRONMENT "BOREALIS_CLIPBOARD=off;QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software;DBUS_SESSION_BUS_ADDRESS=unix:path=/tmp/opal-no-bus;QT_ACCESSIBILITY=0")
